Lemma exact_tail_withdrawal_count_call `{!islaG Sigma} `{!threadG}
    {Decoded} `{!WriteSuccessMemoryRepresentation Decoded}
    (args : WriteSuccessArgs Decoded) (witness : writeSuccessNativeWitness Decoded)
    (P : iProp Sigma) :
  writeSuccessCompleteNativeCode -∗
  instr_body 0x15688 (exactTailWithdrawalCountCallEntry args witness P).
Proof.
  iStartProof. iIntros "#Hnative".
  iEval (rewrite /writeSuccessCompleteNativeCode) in "Hnative".
  iDestruct "Hnative" as "(#Hprologue & #Hlookup & #Hmain & #Houtput &
    #Hmemcpy & #Hboolean & #Hoptional & #Hlists & #Hbytes & #Hint & #Hu256 & #Hu128)".
  iPoseProof "Hlookup" as "#Hlookup15688".
  iEval (rewrite /instr_lookup_resources) in "Hlookup15688".
  iPoseProof ("Hlookup15688" $! 0x15688 with "[]") as "#Hinstr15688".
  { iPureIntro. rewrite IslaTraces.write_success_blocks.Index.lookup_a15688. done. }
  iEval (rewrite IslaTraces.write_success_blocks.Index.lookup_a15688) in "Hinstr15688".
  iApply (exact_tail_withdrawal_count_call_core with "Hinstr15688 Hint Houtput").
Qed.
Print Assumptions exact_tail_withdrawal_count_call.
